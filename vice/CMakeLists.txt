cmake_minimum_required(VERSION 3.25)
project(vice VERSION 0.1.0 LANGUAGES CXX)

option(VICE_BUILD_CLI "Build the ViceScript command line" ON)
option(VICE_BUILD_BENCHMARKS "Build the standalone ViceScript VM benchmarks" OFF)

add_library(vice STATIC value.cpp runtime.cpp collections.cpp library.cpp standard_library.cpp lexer.cpp parser.cpp compiler.cpp vm.cpp)
add_library(vice::vice ALIAS vice)
target_compile_features(vice PUBLIC cxx_std_20)
target_include_directories(vice PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(vice PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(vice PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(vice PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(VICE_BUILD_BENCHMARKS)
    add_executable(vice_benchmarks benchmarks.cpp)
    target_link_libraries(vice_benchmarks PRIVATE vice::vice)
endif()

if(VICE_BUILD_CLI)
    add_executable(vice_cli main.cpp)
    target_link_libraries(vice_cli PRIVATE vice::vice)
    set_target_properties(vice_cli PROPERTIES OUTPUT_NAME vice ARCHIVE_OUTPUT_NAME vice_cli)
endif()
